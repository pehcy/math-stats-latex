\chapter{Large Sample Theory}

Sometimes it is difficult to determine the exact distribution of a statistic for a finite value 
of $n$, and the assumptions made about the probability distribution may not be valid. 
In such cases, if the limiting distributions of the statistics exist, then the problems of testing 
such hypothesis and of setting confidence intervals may be easily solved for large samples.
However, these methods will be only approximate for a finite value of $n$.

On the bright side, these large-sample approximate results can make advantage since 
we do not have to make assumptions about the parent population and that in most cases 
the limiting distribution is normal. Thus normal theory can be applied to get 
approximate tests and confidence intervals. Another important 
limiting distribution used in connection with categorical data is the chi-squared distribution.
Such large-sample methods are beneficial in practical applications.

If the symptotic distribution of a statistic is normal, then for a large $n$ we can 
perform approximate large-sample tests on mean or variance of the statistic. In this chapter, we shall consider 
the case of categorical data and discuss certain transformations of statistics that are widely used in 
large samples.

\section{Transformation of statistics to stabilize variance}

\begin{lemma}
    \label{lem:las01}
    If $\{T_n\}_{n \geq 1}$ is a sequence of statistics such that 
    \begin{equation}
        \sqrt{n}(T_n - \theta) \xrightarrow[]{\mathcal{D}} Z \sim N\left(0, \sigma^2_T(\theta) \right),
    \end{equation}
    then the transformation of statistics $g(T_n)$ formed another sequence that
    \begin{equation}
        \sqrt{n}\left[ g(T_n) - g(\theta)\right] \xrightarrow[]{\mathcal{D}} \widetilde{Z} \sim N\left(0, \left[\sigma^2_T(\theta) g'(\theta) \right]^2 \right),
    \end{equation}
    provided $g(\theta)$ is differentiable and is nonzero.
\end{lemma}
\begin{proof}
    Consider the Taylor expansion of $g(T_n)$ around $g(\theta)$, that is,
    \[
        g(T_n) = g(\theta) + (T_n - \theta) [g'(\theta) + \epsilon_n]
    \]
    where $\epsilon \to 0$ as $T_n \to \theta$. Therefore, for any $\epsilon > 0$, $|\epsilon_n|<\epsilon$ whenerver 
    $|T_n - \theta| < \delta$. Hence, for any $\epsilon$ greater than zero, we have 
    \begin{align*}
        \mathbb{P} \left[ |\epsilon_n| < \epsilon\right] &= \mathbb{P} \left[ |T_n - \theta| < \delta \right]\\
        &= \mathbb{P} \left[ \frac{\sqrt{n} |T_n - \theta|}{\sigma_T(\theta)} < \frac{\sqrt{n} \delta}{\sigma_T(\theta)}\right]\\
        &= \mathbb{P} \left[ |\tau| < \frac{\sqrt{n} \delta}{\sigma_T(\theta)}\right] & \text{where } \tau \sim N(0,1)\\
        &= \left\{2 \Phi \left( \frac{\sqrt{n} \delta}{\sigma_T(\theta)}\right) - 1 \right\} \to 1 \quad \text{as } n \to \infty.
    \end{align*}
    By definition, $\epsilon$ converges to zero in probability. So this yields
    \[
        \sqrt{n}\left[ g(T_n) - g(\theta)\right] - \sqrt{n}(T_n - \theta)g'(\theta) = \fbox{$\sqrt{n}(T_n - \theta)\, \epsilon \xrightarrow[]{p} 0$}.
    \]
    Therefore, 
    \[
        \sqrt{n}\left[ g(T_n) - g(\theta)\right] \xrightarrow[]{\mathcal{D}} \sqrt{n}(T_n - \theta)g'(\theta) \sim N\left(0, \left[\sigma^2_T(\theta) g'(\theta) \right] \right).
    \]
    This completes the proof.
\end{proof}

\section{Stabilization of variance}

\index{Variance stabilization} is a statistical technique to 
make the spread (variance) of data consistent across its range by removing its dependence on the mean, which simplifies 
analysis and improves model performance by creating homoscedasticity (constant variance). It is crucial for data like 
counts (Poisson, Binomial) where variance naturally changes with the average, allowing 
standard methods (like ANOVA, regression) to work effectively and yield more reliable results, especially for small samples. 

In many real-world datasets (such as 
\textit{counts of rare events}, \textit{proportions}), the variability (variance) increases as the average value (mean) increases, 
violating assumptions of many statistical tests. For example,
we may not have enough points to determine if the variance is related to
the mean, or whether the normality assumption is satisfied. In this situa-
tion you may have to guess the appropriate transformation. For count data
you would use the $\sqrt{X}$ or $\ln X$ transformation. Most count data are more
overdispersed or clumped than the Poisson distribution, however, and so the
log X transformation will usually be a better choice than $\sqrt{X}$. You would
use the $\arcsin(\sqrt{X})$ transformation for proportion data, especially if there
are some proportions near 0 or 1.

\begin{theorem}[Variance Stabilizing Transformation]
    If $\{T_n\}_{n \geq 1}$ is a sequence of statistics such that $\sqrt{n}(T_n - \theta) \sim N\left(0, \sigma^2_T(\theta) \right)$, 
    where $\sigma^2_T(\theta)$ is a function depends on $\theta$. Then 
    \begin{equation}
        g(\theta) \varpropto \int \frac{\mathrm{d}\theta}{\sigma_T(\theta)}
    \end{equation}
    is a \textbf{variance stabilizing transformation}.
\end{theorem}
\begin{proof}
    From \cref{lem:las01}, we have 
    \[
        \sqrt{n}\left[ g(T_n) - g(\theta)\right] \xrightarrow[]{\mathcal{D}} \widetilde{Z} \sim N\left(0, \left[\sigma^2_T(\theta) g'(\theta) \right]^2 \right)
    \]
    provided $g'(\theta)$ exists. We wish to find a transformation $g(T_n)$ whose variance is independent of $\theta$ (or, the variance is stable with respect to $\theta$). 
To stabilize the variance, we pick a function $g(T_n)$, such that 
\[
    [\sigma_T(\theta)\, g'(\theta)]^2 = c,
\]
where $c$ is a constant that is independent of $\theta$. The asymptotic variance of the transformed statistic $g(T_n)$ will be independent of $\theta$. 

Now, solving the differential equation $\sigma_T(\theta)\, g'(\theta) = c$.
\begin{align*}
    \sigma_T(\theta)\, g'(\theta) = c &\implies g'(\theta) = \frac{c}{\sigma_T(\theta)}\\
    &\implies \int g'(\theta) \> \mathrm{d}\theta = c \int \frac{\mathrm{d}\theta}{\sigma_T(\theta)}\\
    &\implies g(\theta) = c \int \frac{\mathrm{d}\theta}{\sigma_T(\theta)}.
\end{align*}
$g(\theta)$ is now proportional to the integral of the reciprocal of standard deviation.
\end{proof}

From the proof we know that $\sqrt{n}\left[ g(T_n) - g(\theta)\right]$ is asymptotically normal with mean $0$ and 
variance $c^2$, and that 
\begin{equation}
    \frac{\sqrt{n}\left[ g(T_n) - g(\theta)\right]}{c} \sim AN(0,1).
\end{equation}
When testing $H_0 : \theta = \theta_0$ versus $H_0 : \theta \neq \theta_0$ at significance
level $\alpha$, we reject $H_0$ if 
\begin{equation}
    \left| \frac{\sqrt{n}\left[ g(T_n) - g(\theta)\right]}{c} \right| > z_{\alpha/2}.
\end{equation}

\section{Approximate Tests and Confidence Intervals for proportion}

A variance stabilizing transformation is important for proportions,  
this is because the variance of a proportion depends on its mean. To 
see this, imagine that we have observed $n$ different individuals 
from a population and record their sex. Let $X$ be the number of 
individuals in the sample that are female. The 
variable $X$ would be a binomial random variable with 
parameters $n$ and $p$, where $p$ is the proportion of females in the population,
thus $\mathbb{E} X = np$ and $\text{Var}(X) = np(1-p)$. Then 
a \textbf{binomial proportion} of females would be $p = \frac{X}{n}$.
For the binomial proportion, we have
\begin{equation}
    \mathbb{E}(\widehat{p}) = p, \quad \text{Var}(\widehat{p}) = \frac{p(1-p)}{n}.
\end{equation}

Using the stabilization of variance transformation, the variance-stabilized binomial 
proportions is $\sin^{-1}(\sqrt{X})$ or $\arcsin(\sqrt{X})$. This transformation maps 
proportions from 0 to 1 into angles from 0 to $\pi/2$ radians. The largest 
effect of the transformations is on proportions near 0 or 1.

In general, by CLT, we have 
\begin{equation}
    \frac{\widehat{p} - \mathbb{E}(\widehat{p})}{\sqrt{\text{Var}(\widehat{p})}} \sim N(0,1).
\end{equation}
Our interest is to test the hypothesis $H_0 : p = p_0$ versus $H_1 : p \neq p_0$ at significance level $\alpha$.
The test statistic should be 
\begin{equation}
    Z = \frac{\widehat{p} - p_0}{\sqrt{\text{Var}(\widehat{p})}} = \frac{\widehat{p} - p_0}{\sqrt{\frac{p_0(1-p_0)}{n}}}.
\end{equation}
We reject $H_0$ if $|Z| > z_{\alpha/2}$, where $z_{\alpha/2}$ is the upper $\alpha/2$ quantile of the standard normal distribution.

For large $n$, the estimate of $\sigma^2$ is 
\begin{equation}
    \widehat{\sigma}^2_{\widehat{p}} = \frac{\widehat{p} (1 - \widehat{p})}{n} 
    = \frac{\sqrt{n} (\widehat{p} - p_0)}{\sqrt{\widehat{p}\, (1-\widehat{p})}}
    \sim N(0,1).
\end{equation}
An approximate $100(1-\alpha)\%$ confidence interval for $p$ is
\begin{equation}
    \mathbb{P} \left[ \left \vert \frac{\sqrt{n} (\widehat{p} - p_0)}{\sqrt{\widehat{p}\, (1-\widehat{p})}} \right \vert \leq z_{\alpha/2} \right] 
    = 1 - \alpha.
\end{equation}
Hence, solving the inequality to get the confidence interval for $p$ as
\begin{align*}
    &\mathbb{P} \left[ -z_{\alpha/2} \leq \frac{\sqrt{n} (\widehat{p} - p_0)}{\sqrt{\widehat{p}\, (1-\widehat{p})}} \leq z_{\alpha/2} \right] = 1 - \alpha\\
    &\Longleftrightarrow \mathbb{P} \left[ -z_{\alpha/2} \sqrt{\frac{\widehat{p}\, (1-\widehat{p})}{n}} \leq (\widehat{p} - p_0) \leq z_{\alpha/2} \sqrt{\frac{\widehat{p}\, (1-\widehat{p})}{n}} \right] = 1 - \alpha\\
    &\Longleftrightarrow \mathbb{P} \left[ -\widehat{p} -z_{\alpha/2} \sqrt{\frac{\widehat{p}\, (1-\widehat{p})}{n}} \leq -p_0 \leq -\widehat{p} + z_{\alpha/2} \sqrt{\frac{\widehat{p}\, (1-\widehat{p})}{n}} \right] = 1 - \alpha\\
    &\Longleftrightarrow \mathbb{P} \left[ \widehat{p} -z_{\alpha/2} \sqrt{\frac{\widehat{p}\, (1-\widehat{p})}{n}} \leq p_0 \leq \widehat{p} + z_{\alpha/2} \sqrt{\frac{\widehat{p}\, (1-\widehat{p})}{n}} \right] = 1 - \alpha\\
\end{align*}

We now applying arcsine transformation to stabilize the variance of $\widehat{p}$, recall that 
\begin{equation}
    \arcsin \sqrt{\widehat{p}} \sim N\left(\arcsin \sqrt{p}, \frac{1}{4n}\right).
\end{equation}
An approximate $100(1-\alpha)\%$ confidence interval for $p$ is
\begin{align*}
    &\mathbb{P} \left[ \arcsin \sqrt{\widehat{p}} - z_{\alpha / 2} \sqrt{\frac{1}{4n}} \leq \arcsin \sqrt{p} \leq \arcsin \sqrt{\widehat{p}} + z_{\alpha / 2} \sqrt{\frac{1}{4n}} \right] = 1 - \alpha\\
    &\Longleftrightarrow \mathbb{P} \left[ \sin^2 \left( \arcsin \sqrt{\widehat{p}} - z_{\alpha / 2} \sqrt{\frac{1}{4n}} \right) \leq p \leq \sin^2 \left( \arcsin \sqrt{\widehat{p}} + z_{\alpha / 2} \sqrt{\frac{1}{4n}} \right) \right] = 1 - \alpha\\
\end{align*}

\begin{remark}
    In trigonometry, we write $\sin^2 x = (\sin x)^2$.
\end{remark}

\begin{example}
    Consider a Poisson distribution with mean $\lambda$. Let an 
    estimator of $\lambda$ be 
    \[
        \widehat{\sqrt{\lambda}} = \frac{n_1 \sqrt{\ols{X_1}} + n_2 \sqrt{\ols{X_2}}}{n_1 + n_2},
    \]
    Construct a $100(1-\alpha)\%$ confidence interval for $\sqrt{\lambda}$.
\end{example}
\begin{solution}
    By CLT, $\displaystyle \frac{\widehat{\sqrt{\lambda}} - \sqrt{\lambda}}{\sqrt{\text{Var}(\widehat{\sqrt{\lambda}})}} \sim N(0,1)$

    An $100(1 - \alpha)\%$ confidence interval for $\sqrt{\lambda}$ is
    \begin{align*}
        &\mathbb{P} \left[ -z_{\alpha/2} \leq \frac{\widehat{\sqrt{\lambda}} - \sqrt{\lambda}}{\sqrt{\text{Var}(\widehat{\sqrt{\lambda}})}} \leq z_{\alpha/2} \right] = 1 - \alpha\\
        &\Longleftrightarrow \mathbb{P} \left[ -z_{\alpha/2} \sqrt{\text{Var}(\widehat{\sqrt{\lambda}})} \leq \widehat{\sqrt{\lambda}} - \sqrt{\lambda} \leq z_{\alpha/2} \sqrt{\text{Var}(\widehat{\sqrt{\lambda}})} \right] = 1 - \alpha\\
        &\Longleftrightarrow \mathbb{P} \left[ \widehat{\sqrt{\lambda}} - z_{\alpha/2} \sqrt{\text{Var}(\widehat{\sqrt{\lambda}})} \leq \sqrt   {\lambda} \leq \widehat{\sqrt{\lambda}} + z_{\alpha/2} \sqrt{\text{Var}(\widehat{\sqrt{\lambda}})} \right] = 1 - \alpha
    \end{align*}

    Compute $\text{Var}(\widehat{\sqrt{\lambda}})$:
    \begin{align*}
        \text{Var}(\widehat{\sqrt{\lambda}}) &= \text{Var}\left( \frac{n_1 \sqrt{\ols{X_1}} + n_2 \sqrt{\ols{X_2}}}{n_1 + n_2} \right)\\
        &= \frac{n_1^2}{(n_1 + n_2)^2} \text{Var}(\sqrt{\ols{X_1}}) + \frac{n_2^2}{(n_1 + n_2)^2} \text{Var}(\sqrt{\ols{X_2}})\\
        &= \frac{n_1^2}{(n_1 + n_2)^2} \frac{1}{4 n_1} + \frac{n_2^2}{(n_1 + n_2)^2} \frac{1}{4 n_2}\\
        &= \frac{1}{4 (n_1 + n_2)}
    \end{align*}
\end{solution}
